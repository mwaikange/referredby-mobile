### ✅ REPLACE WITH THIS CODE:

```typescript
const fetchPaymentHistory = async () => {
  try {
    setLoading(true);
    
    // Get the current user ID (however you're getting it in your app)
    const userId = profile?.id; // or however you access the user ID
    
    if (!userId) {
      console.error("No user ID found");
      setRecords([]);
      setLoading(false);
      return;
    }

    console.log("Fetching payment history for user:", userId);

    // Call the NEW API endpoint
    const response = await fetch(
      `https://appv2.referredby.com.na/api/mobile/payment-history?user_id=${userId}`
    );

    if (!response.ok) {
      throw new Error(`API error: ${response.status}`);
    }

    const data = await response.json();
    
    console.log("Payment history response:", data);

    // The API returns { success: true, payments: [...] }
    if (data.success && data.payments && data.payments.length > 0) {
      setRecords(data.payments);
      console.log("Loaded", data.payments.length, "payment records");
    } else {
      setRecords([]);
      console.log("No payment records found");
    }

  } catch (error) {
    console.error("Error fetching payment history:", error);
    setRecords([]);
  } finally {
    setLoading(false);
  }
};
```
