async function handleLogin(email: string, pin: string) {
  console.log('----------------------------------------');
  console.log('🔐 ATTEMPTING LOGIN');
  console.log('----------------------------------------');
  console.log('📧 Email:', email);
  
  try {
    // Step 1: Authenticate with Supabase
    console.log('🔑 Signing in with Supabase...');
    
    const { data, error } = await supabase.auth.signInWithPassword({
      email: email,
      password: pin,
    });

    if (error) {
      console.error('❌ Authentication failed:', error.message);
      alert(`Login failed: ${error.message}`);
      return;
    }

    if (!data.session) {
      console.error('❌ No session created');
      alert('Login failed: No session returned');
      return;
    }

    console.log('✅ Authentication successful!');
    console.log('👤 User ID:', data.user.id);

    // Step 2: Fetch user profile from backend API (UPDATED!)
    console.log('📥 Fetching profile from API...');
    
    const profileResponse = await fetch(`${ENV.API_BASE_URL}/api/users/me`, {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${data.session.access_token}`,
        'Content-Type': 'application/json',
      },
    });

    console.log('📡 Profile response status:', profileResponse.status);

    if (!profileResponse.ok) {
      const errorText = await profileResponse.text();
      console.error('❌ Profile fetch failed:', errorText);
      alert('Could not load profile data');
      return;
    }

    const profileData = await profileResponse.json();
    
    console.log('✅ Profile loaded from API!');
    console.log('👤 Name:', profileData.first_name, profileData.last_name);
    console.log('🆔 UID:', profileData.uid);
    console.log('📱 Mobile:', profileData.mobile);
    console.log('💰 Nano Limit:', profileData.nano_loan_limit);
    console.log('💰 Term Limit:', profileData.term_loan_limit);
    console.log('⭐ Rating:', profileData.borrower_rating);

    // Step 3: Navigate to profile page
    console.log('🎯 Navigating to Profile...');
    navigation.navigate('Profile', { 
      user: profileData,
      session: data.session,
    });
    
    console.log('✅ Login complete!');
    console.log('----------------------------------------');

  } catch (error) {
    console.error('❌ ERROR:', error.message);
    alert(`Login failed: ${error.message}`);
  }
}

const InterestConfirmationPage = ({ route }) => {
  const { user, session } = route.params;
  const [loading, setLoading] = useState(true);
  const [data, setData] = useState(null);
  const [error, setError] = useState(null);
  
  useEffect(() => {
    fetchInterestData();
  }, []);
  
  const fetchInterestData = async () => {
    try {
      console.log('📥 Fetching interest confirmation from API...');
      
      const response = await fetch(
        `${ENV.API_BASE_URL}/api/mobile/interest-confirmation`,
        {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${session.access_token}`,
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            user_id: user.id,
            loan_amount: user.nano_loan_limit || 2000,
          }),
        }
      );
      
      console.log('📡 API Response Status:', response.status);
      
      if (!response.ok) {
        throw new Error(`API Error: ${response.status}`);
      }
      
      const apiData = await response.json();
      console.log('✅ Interest data received:', apiData);
      
      setData(apiData);
      setLoading(false);
      
    } catch (err) {
      console.error('❌ Interest API Error:', err.message);
      setError(err.message);
      setLoading(false);
    }
  };
  
  if (loading) {
    return (
      <View style={styles.loadingContainer}>
        <Text>Loading interest rates...</Text>
      </View>
    );
  }
  
  if (error) {
    return (
      <View style={styles.errorContainer}>
        <Text style={styles.errorTitle}>Unable to Load Interest Rates</Text>

          style={styles.retryButton}

          <Text style={styles.label}>Lender</Text>
          <Text style={styles.value}>{data.portfolio?.full_name || 'N/A'}</Text>
        </View>
        <View style={styles.infoRow}>
          <Text style={styles.label}>Lending Society</Text>
          <Text style={styles.value}>{data.lending_society?.name || 'N/A'}</Text>
        </View>
        <View style={styles.infoRow}>
          <Text style={styles.label}>Borrower</Text>
          <Text style={styles.value}>{user.first_name} {user.last_name}</Text>
        </View>
      </View>

      <View style={styles.rateBadge}>
        <Text style={styles.rateBadgeText}>
          Active Interest Mode: {data.interest?.rate_basis || 'IIR'} (Rating-Based)
