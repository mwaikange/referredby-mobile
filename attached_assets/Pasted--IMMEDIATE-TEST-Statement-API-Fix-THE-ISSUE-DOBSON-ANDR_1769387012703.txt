Future<void> fetchStatement() async {
  final userId = await getUserId(); // Get from secure storage/session

  );
  
  if (response.statusCode == 200) {
    final data = jsonDecode(response.body);
    
    if (data['success'] == true && data['loan'] != null) {
      // Display the loan
      setState(() {
        loanId = data['loan']['loan_id'];
        loanType = data['loan_type']; // 'nano' or 'term'
        loanAmount = data['loan']['loan_amount'];
        interestRate = data['loan']['interest_rate'];
        interestAmount = data['loan']['interest_amount'];
        processingFee = data['loan']['processing_fee_amount'];
        totalRepayable = data['loan']['total_repayable'];
        dueDate = data['loan']['due_date'];
        outstandingDate = data['loan']['outstanding_date'];
        isActive = data['is_active'];
        isPaidUp = data['is_paid_up'];
      });
    } else {
      // No loans found - show "Apply for loan" screen
      showNoLoansScreen();
    }
  }
}
